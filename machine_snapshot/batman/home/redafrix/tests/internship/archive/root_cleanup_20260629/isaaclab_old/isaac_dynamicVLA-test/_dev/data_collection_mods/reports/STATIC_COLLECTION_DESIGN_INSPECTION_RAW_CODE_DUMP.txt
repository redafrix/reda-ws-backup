    69	    categories = sorted([f for f in os.listdir(object_dir)])
    70	    for c in categories:

   367	def _get_object_range_bbox(table_bbox, robot_position=None, robot_reach_dist=None):
   368	    from pxr import Gf
   369	
   370	    object_range_min_0 = table_bbox.min[0] * 3 / 4 + table_bbox.max[0] / 4
   371	    object_range_max_0 = table_bbox.min[0] / 4 + table_bbox.max[0] * 3 / 4
   372	    object_range_min_1 = table_bbox.min[1] * 3 / 4 + table_bbox.max[1] / 4
   373	    object_range_max_1 = table_bbox.min[1] / 4 + table_bbox.max[1] * 3 / 4
   374	    table_z = table_bbox.max[2]
   375	    object_valid_range = Gf.Range3d(
   376	        Gf.Vec3d(object_range_min_0, object_range_min_1, table_z),
   377	        Gf.Vec3d(object_range_max_0, object_range_max_1, table_z),
   378	    )
   379	    if robot_position is None and robot_reach_dist is None:
   380	        return object_valid_range
   381	    else:
   382	        # Consider whether the object is within the robot reach
   383	        robot_reach_bbox = Gf.Range3d(
   384	            Gf.Vec3d(
   385	                robot_position[0] - robot_reach_dist,
   386	                table_bbox.min[1] - robot_reach_dist,
   387	                table_z,
   388	            ),
   389	            Gf.Vec3d(
   390	                robot_position[0] + robot_reach_dist,
   391	                table_bbox.min[1] + robot_reach_dist,
   392	                table_z,
   393	            ),
   394	        )
   395	        return robot_reach_bbox.IntersectWith(object_valid_range)

   450	def _get_dynamic_object_state(
   451	    object_range_bbox, object_z, moving_speed, friction, robot_position
   452	):
   453	    import configs.object_cfg
   454	
   455	    object_position = np.array(
   456	        [
   457	            random.uniform(object_range_bbox.min[0], object_range_bbox.max[0]),
   458	            random.uniform(object_range_bbox.min[1], object_range_bbox.max[1]),
   459	            object_z,
   460	        ]
   461	    )
   462	    assert robot_position is not None
   463	    # Generate a random position between the table center and the robot arm
   464	    tbl_ctr = (object_range_bbox.min + object_range_bbox.max) / 2.0
   465	    random_ratio = random.uniform(-0.5, 0.5)
   466	    random_position = tbl_ctr + random_ratio * (robot_position - tbl_ctr)
   467	    random_position[2] = object_z
   468	    # Determine the linear velocity of the object
   469	    assert moving_speed is not None and len(moving_speed) == 2
   470	    object_direction = random_position - object_position
   471	    object_velocity = (
   472	        object_direction
   473	        / np.linalg.norm(object_direction)
   474	        * random.uniform(*moving_speed)
   475	    )
   476	    object_quat = configs.object_cfg.get_object_init_quat(object_velocity)
   477	
   478	    return {
   479	        "pos": object_position,
   480	        "quat": object_quat,
   481	        "lin_vel": object_velocity,
   482	        "friction": friction,
   483	    }

   487	    container_size, object_range_bbox, random_orientation, existing_objects
   488	):
   489	    import configs.object_cfg
   490	
   491	    N_MAX_TRIES = 100
   492	    n_tries = 0
   493	    container_position = None
   494	    while container_position is None and n_tries < N_MAX_TRIES:
   495	        n_tries += 1
   496	        container_position = np.array(
   497	            [
   498	                random.uniform(object_range_bbox.min[0], object_range_bbox.max[0]),
   499	                random.uniform(object_range_bbox.min[1], object_range_bbox.max[1]),
   500	                _get_object_z(object_range_bbox.max[2], container_size),
   501	            ]
   502	        )
   503	        container_quat = np.array([1.0, 0.0, 0.0, 0.0])
   504	        if random_orientation:
   505	            container_quat = configs.object_cfg.get_object_init_quat(
   506	                np.random.uniform(-0.1, 0.1, size=3), upright=True
   507	            )
   508	        # Check whether the container is occluding with existing objects/containers
   509	        for eo in existing_objects["objects"] + existing_objects["containers"]:
   510	            if _is_bbox_overlap(
   511	                _get_object_bbox(container_position, container_size, container_quat),
   512	                _get_object_bbox(
   513	                    eo["pos"], eo["size"], eo["quat"], eo.get("lin_vel", None)
   514	                ),
   515	            ):
   516	                container_position = None
   517	                break
   518	
   519	    if container_position is not None:
   520	        return {"pos": container_position, "quat": container_quat}
   521	    else:
   522	        return None

   525	def _get_object_bbox(position, size, quat, lin_vel=None):
   526	    # TODO: Consider the velocity of the object
   527	    dx, dy, dz = size / 2.0
   528	    corners = np.array(
   529	        [
   530	            [-dx, -dy, -dz],
   531	            [dx, -dy, -dz],
   532	            [dx, dy, -dz],
   533	            [-dx, dy, -dz],
   534	            [-dx, -dy, dz],
   535	            [dx, -dy, dz],
   536	            [dx, dy, dz],
   537	            [-dx, dy, dz],
   538	        ]
   539	    )
   540	    rotated_corners = R.from_quat(quat[[1, 2, 3, 0]]).apply(corners)
   541	
   542	    bbox = rotated_corners + position
   543	    return Polygon(bbox[np.argsort(bbox[:, 2])[:4], :2])

   645	    if unsqueeze:

   794	def get_next_object(scene_objects, scene, env_idx=None):
   795	    next_object = []  # next object index for each environment
   796	    n_envs = len(scene_objects)
   797	    for i in range(n_envs):
   798	        if len(scene_objects[i]) == 0 or (env_idx is not None and i != env_idx):
   799	            next_object.append(None)
   800	            continue
   801	
   802	        objects = scene_objects[i]
   803	        objects_velocity = torch.cat(
   804	            [scene[o].data.root_lin_vel_w[i : i + 1] for o in objects], dim=0
   805	        )
   806	        speed = torch.norm(objects_velocity, dim=-1)
   807	        fastest_index = torch.argmax(speed, dim=0).item()
   808	        next_object.append(fastest_index)
   809	
   810	    return next_object if env_idx is None else next_object[env_idx]

   813	def get_env_states(states, n_envs=1):
   814	    KEYS = {
   815	        "sm_state": ("next_state", "sm_state"),
   816	        "ee_pos": ("curr_state", "end_effector", "pos"),
   817	        "ee_quat": ("curr_state", "end_effector", "quat"),
   818	        "joints": ("curr_state", "joints"),
   819	        "object_pos": ("curr_state", "object", "pos"),
   820	        "object_quat": ("curr_state", "object", "quat"),
   821	        "object_vel": ("curr_state", "object", "velocity"),
   822	        "action": ("next_state", "action"),
   823	    }
   824	    if not states:
   825	        return []
   826	
   827	    env_states = [{} for _ in range(n_envs)]
   828	    for es in states:
   829	        # es = {"cam_views": dict, "curr_state": dict, "next_state": dict, "is_done": Tensor}
   830	        for eid in range(n_envs):
   831	            # Predefined keys
   832	            for k, v in KEYS.items():
   833	                value = None
   834	                if v[0] not in es:
   835	                    continue
   836	                # Retrieve the value based on the key path
   837	                if len(v) == 1:
   838	                    value = es[v[0]][eid].cpu().numpy()
   839	                elif len(v) == 2 and v[1] in es[v[0]]:
   840	                    value = es[v[0]][v[1]][eid].cpu().numpy()
   841	                elif len(v) == 3 and v[1] in es[v[0]] and v[2] in es[v[0]][v[1]]:
   842	                    value = es[v[0]][v[1]][v[2]][eid].cpu().numpy()
   843	
   844	                if value is None:
   845	                    continue
   846	                if k not in env_states[eid]:
   847	                    env_states[eid][k] = []
   848	
   849	                env_states[eid][k].append(value)
   850	
   851	            # Camera views
   852	            if "cam_views" in es:
   853	                for cam, imgs in es["cam_views"].items():
   854	                    for k, v in imgs.items():
   855	                        cam_key = "%s_%s" % (cam, k)
   856	                        if cam_key not in env_states[eid]:
   857	                            env_states[eid][cam_key] = []
   858	
   859	                        env_states[eid][cam_key].append(v[eid])
   860	
   861	    return env_states

   864	def simulate(sim_cfg, task, robot, scene_dir, object_metadata, seed):
   865	    import configs.termination_cfg
   866	
   867	    # Create a new environment
   868	    env_cfg, object_tags, objects, object_sizes = get_env_cfg(
   869	        sim_cfg,
   870	        task,
   871	        robot,
   872	        object_metadata,
   873	        scene_dir,
   874	    )
   875	    # Check whether the object tags are empty
   876	    if not object_tags["objects"]:
   877	        raise ValueError("No object tags found. Skipping simulation.")
   878	    if (
   879	        sim_cfg["scene"]["containers"]["n_containers"] > 0
   880	        and not object_tags["containers"]
   881	    ):
   882	        raise ValueError("No container tags found. Skipping simulation.")
   883	
   884	    env = gym.make("Robot-Env-Cfg-v0", cfg=env_cfg, seed=seed)
   885	    # Reset environment at start
   886	    env.reset(seed=seed)
   887	
   888	    # Determine the object size (without transformation)
   889	    if "container" in env.unwrapped.scene.keys():
   890	        container_data = env.unwrapped.scene["container"].data
   891	        container_size = get_object_size(
   892	            os.path.basename(env_cfg.scene.container.spawn.usd_path),
   893	            object_metadata,
   894	            env.unwrapped.device,
   895	        )
   896	    else:
   897	        container_data, container_size = None, None
   898	
   899	    # Enable Path Tracing
   900	    if sim_cfg["enable_cameras"] and sim_cfg["path_tracing"]:
   901	        import omni.replicator.core as rep
   902	
   903	        rep.settings.set_render_pathtraced()
   904	
   905	    # Initialize the state machine
   906	    assert task in sim_cfg["tasks"], "Unknown task: %s." % task
   907	    assert robot in sim_cfg["robots"], "Unknown robot: %s." % robot
   908	    state_machine = get_state_machine(
   909	        sim_cfg["tasks"][task],
   910	        sim_cfg["robots"][robot],
   911	        {
   912	            "dt": env_cfg.sim.dt * env_cfg.decimation,
   913	            "num_envs": env.unwrapped.num_envs,
   914	            "device": env.unwrapped.device,
   915	        },
   916	    )
   917	
   918	    # Set the rotation, height, and physical material of the object
   919	    set_object_material(
   920	        env.unwrapped.scene["object"],
   921	        n_envs=env.unwrapped.num_envs,
   922	    )
   923	
   924	    # Simulation loop
   925	    env_states = []
   926	    term_mgr = env.env.termination_manager

  1007	def is_object_stopped(scene_cfg, object_velocity, n_steps=25):
  1008	    init_speed = np.linalg.norm(scene_cfg["object"]["init_state"]["lin_vel"])
  1009	    # The object is stopped within n_steps
  1010	    for i in range(min(n_steps, len(object_velocity))):
  1011	        if init_speed > 1e-2 and np.linalg.norm(object_velocity[i]) < 1e-2:

  1060	def get_object_without_numpy(obj):
  1061	    if obj is None:
  1062	        return obj
  1063	    elif isinstance(obj, (str, int, float, bool)):
  1064	        return obj
  1065	    elif isinstance(obj, dict):
  1066	        return {k: get_object_without_numpy(v) for k, v in obj.items()}
  1067	    elif isinstance(obj, list):
  1068	        return [get_object_without_numpy(item) for item in obj]
  1069	    elif isinstance(obj, tuple):
  1070	        return tuple(get_object_without_numpy(item) for item in obj)
  1071	    elif isinstance(obj, torch.Tensor):
  1072	        return obj.cpu().numpy().tolist()
  1073	    elif isinstance(obj, np.ndarray):
  1074	        return obj.tolist()
  1075	    elif isinstance(obj, (np.int32, np.float64)):
  1076	        return obj.item()
  1077	    else:
  1078	        # logging.warning("Unknown data type: %s" % type(obj))
  1079	        return str(obj)

  1082	def get_frames(
  1083	    env_state, state_keys=["sm_state", "ee_pos", "object_pos", "object_vel"]
  1084	):
  1085	    MAX_DEPTH = 25
  1086	
  1087	    cam_name = None
  1088	    cam_frames = {}
  1089	    for st_key, frames in env_state.items():
  1090	        cam_idx = st_key.find("_cam_")
  1091	        if cam_idx == -1:
  1092	            continue
  1093	
  1094	        cam_name = st_key[:cam_idx]
  1095	        img_name = st_key[cam_idx + 5 :]
  1096	        if cam_name not in cam_frames:
  1097	            cam_frames[cam_name] = {}
  1098	        if img_name not in cam_frames[cam_name]:
  1099	            cam_frames[cam_name][img_name] = []
  1100	
  1101	        for frame in frames:
  1102	            if frame.ndim == 2:
  1103	                frame = np.repeat(frame[:, :, None], 3, axis=-1)
  1104	            elif frame.ndim == 3:
  1105	                if frame.shape[-1] == 1:
  1106	                    frame = np.repeat(frame, 3, axis=-1)
  1107	                elif frame.shape[-1] >= 3:
  1108	                    frame = frame[:, :, :3]
  1109	            else:
  1110	                raise ValueError("Unknown camera data shape: %s" % (frame.shape,))
  1111	
  1112	            # Normalize the depth image to 0-255
  1113	            if img_name in ["depth", "distance_to_image_plane", "distance_to_camera"]:
  1114	                frame = np.clip(frame, 0, MAX_DEPTH)
  1115	                frame = (frame / np.max(frame) * 255).astype(np.uint8)
  1116	            if img_name in ["seg", "semantic_segmentation"]:
  1117	                # Assign a color to each semantic class
  1118	                frame = helpers.get_semantic_map(frame[..., 0])
  1119	
  1120	            cam_frames[cam_name][img_name].append(frame)
  1121	
  1122	    if cam_name is None:
  1123	        raise ValueError("No camera frames found in the environment state.")
  1124	
  1125	    n_frames = len(cam_frames[cam_name][img_name])
  1126	    frames = [[[] for _ in range(len(cam_frames))] for _ in range(n_frames)]
  1127	    for cam_idx, cam_imgs in enumerate(cam_frames.values()):
  1128	        for img in cam_imgs.values():
  1129	            for frame_idx in range(n_frames):
  1130	                frames[frame_idx][cam_idx].append(img[frame_idx])
  1131	
  1132	    for frame_idx in range(n_frames):
  1133	        frame = np.concatenate(
  1134	            [np.concatenate(r, axis=0) for r in frames[frame_idx]], axis=1
  1135	        )
  1136	        if state_keys:
  1137	            frame = _print_state_on_frame(
  1138	                frame,
  1139	                {k: env_state[k][frame_idx] for k in state_keys if k in env_state},
  1140	            )
  1141	
  1142	        frames[frame_idx] = frame
  1143	
  1144	    return frames

  1147	def _print_state_on_frame(frame, state):
  1148	    TEXT_MARGIN = 10
  1149	    TEXT_SCALE = 0.5
  1150	    TEXT_THICKNESS = 1
  1151	    TEXT_COLOR = (255, 255, 255)
  1152	    TEXT_FONT = cv2.FONT_HERSHEY_SIMPLEX
  1153	
  1154	    lines = _get_state_text(state).split("\n")
  1155	    _, img_width = frame.shape[:2]
  1156	    # Print the text on the image
  1157	    y = TEXT_MARGIN
  1158	    for line in lines:
  1159	        (text_width, text_height), _ = cv2.getTextSize(
  1160	            line, TEXT_FONT, TEXT_SCALE, TEXT_THICKNESS
  1161	        )
  1162	        x = img_width - text_width - TEXT_MARGIN
  1163	        frame = cv2.putText(
  1164	            np.ascontiguousarray(frame),
  1165	            line,
  1166	            (x, y + text_height),
  1167	            TEXT_FONT,
  1168	            TEXT_SCALE,
  1169	            TEXT_COLOR,
  1170	            TEXT_THICKNESS,
  1171	            cv2.LINE_AA,
  1172	        )
  1173	        y += text_height + TEXT_MARGIN
  1174	
  1175	    return frame

  1178	def _get_state_text(state):
  1179	    text = ""
  1180	    for k, v in state.items():
  1181	        k = k.replace("_", " ").title()
  1182	        if isinstance(v, (int, np.int32, np.int64)) or v.ndim == 0:
  1183	            text += "%s: %d\n" % (k, v)
  1184	        elif isinstance(v, np.ndarray):
  1185	            # Convert all quaternions to Euler angles
  1186	            if k.find("Quat") != -1:
  1187	                k = k.replace("Quat", "Rot")
  1188	                v = R.from_quat(v).as_euler("xyz", degrees=True)
  1189	
  1190	            text += "%s: " % k
  1191	            text += " ".join(["%.3f" % i for i in v]) + "\n"
  1192	        else:
  1193	            raise ValueError("Unknown State Value Type: %s" % (type(v),))
  1194	
  1195	    return text

    18	def apply_external_force_torque_xy(
    19	    env: ManagerBasedEnv,
    20	    env_ids: torch.Tensor,
    21	    force_range: tuple[float, float],
    22	    torque_range: tuple[float, float],
    23	    asset_cfg: SceneEntityCfg,
    24	    robot_cfg: SceneEntityCfg = SceneEntityCfg("robot"),
    25	):
    26	    asset = env.scene[asset_cfg.name]
    27	    asset_position = asset.data.root_pos_w
    28	    robot_position = env.scene[robot_cfg.name].data.root_pos_w
    29	    is_lifted = abs(robot_position[:, 2] - asset_position[:, 2]) > 0.05
    30	
    31	    if env_ids is None:
    32	        env_ids = torch.arange(env.scene.num_envs, device=asset.device)
    33	
    34	    num_bodies = (
    35	        len(asset_cfg.body_ids)
    36	        if isinstance(asset_cfg.body_ids, list)
    37	        else asset.num_bodies
    38	    )
    39	    size = (len(env_ids), num_bodies, 3)
    40	    random_fdir = math_utils.sample_uniform(-1.0, 1.0, size, asset.device).sign()
    41	    forces = math_utils.sample_uniform(*force_range, size, asset.device) * random_fdir
    42	    random_tdir = math_utils.sample_uniform(-1.0, 1.0, size, asset.device).sign()
    43	    torques = math_utils.sample_uniform(*torque_range, size, asset.device) * random_tdir
    44	    forces[:, :, 2] = 0.0
    45	    torques[:, :, 0] = 0.0
    46	
    47	    # set the forces and torques into the buffers
    48	    asset.set_external_force_and_torque(
    49	        forces, torques, env_ids=env_ids[~is_lifted], body_ids=asset_cfg.body_ids
    50	    )

    82	def get_object_init_quat(
    83	    init_lin_vel: list[float], upright=False, perturbation=None
    84	) -> list[float]:
    85	
    86	    lin_vel_angle = np.arctan2(init_lin_vel[1], init_lin_vel[0])
    87	    verticle_angle = np.pi / 2 * np.random.choice([-1, 1])
    88	    if perturbation is not None:
    89	        lin_vel_angle += np.deg2rad(perturbation)
    90	        verticle_angle += np.deg2rad(perturbation)
    91	
    92	    quat = scipy.spatial.transform.Rotation.from_euler(
    93	        "xyz",
    94	        [
    95	            0 if upright else verticle_angle,
    96	            0,
    97	            lin_vel_angle,
    98	        ],
    99	    ).as_quat()
   100	    return quat[[3, 0, 1, 2]]

   126	def get_quat_from_look_at(cam_pos, cam_look_at):
   127	    fwd_vec = np.array(
   128	        [
   129	            cam_look_at[0] - cam_pos[0],
   130	            cam_look_at[1] - cam_pos[1],
   131	            cam_look_at[2] - cam_pos[2],
   132	        ]
   133	    )
   134	    fwd_vec /= np.linalg.norm(fwd_vec)
   135	    up_vec = np.array([0, 0, 1])
   136	    right_vec = np.cross(up_vec, fwd_vec)
   137	    right_vec /= np.linalg.norm(right_vec)
   138	    up_vec = np.cross(fwd_vec, right_vec)
   139	    R = np.stack([fwd_vec, right_vec, up_vec], axis=1)
   140	    quat = scipy.spatial.transform.Rotation.from_matrix(R).as_quat()
   141	    return [quat[3], quat[0], quat[1], quat[2]]

    21	def is_object_picked(
    22	    env: ManagerBasedRLEnv,
    23	    goal_position: torch.Tensor,
    24	    tolerance: float,
    25	    objects: list[str] = ["object"],
    26	    ee_frame_cfg: SceneEntityCfg = SceneEntityCfg("ee_frame"),
    27	    robot_cfg: SceneEntityCfg = SceneEntityCfg("robot"),
    28	) -> torch.Tensor:
    29	    assert len(objects) == 1, "Only single object picking is supported."
    30	    object = env.scene[objects[0]]
    31	    ee_frame = env.scene[ee_frame_cfg.name]
    32	    robot = env.scene[robot_cfg.name]
    33	
    34	    object_position_w = object.data.root_pos_w
    35	    eef_position_w = ee_frame.data.target_pos_w[..., 0, :]
    36	
    37	    object_eef_dist = torch.norm(eef_position_w - object_position_w, dim=1)
    38	    goal_position_r = goal_position.to(device=robot.data.root_pos_w.device)
    39	    eef_position_r = helpers.get_robot_relative_position(
    40	        ee_frame.data.target_pos_w[..., 0, :] - robot.data.root_pos_w,
    41	        robot.data.root_quat_w,
    42	    )
    43	    eef_goal_dist = torch.norm(goal_position_r - eef_position_r, dim=1)
    44	    return object_eef_dist < tolerance and eef_goal_dist < tolerance

    47	def are_objects_placed(
    48	    env: ManagerBasedRLEnv,
    49	    goal_position: torch.Tensor,
    50	    objects: list[str],
    51	    object_sizes: Dict[str, torch.Tensor],
    52	    container_size: torch.Tensor,
    53	    tolerance: float,
    54	    container_cfg: SceneEntityCfg = SceneEntityCfg("container"),
    55	    ee_frame_cfg: SceneEntityCfg = SceneEntityCfg("ee_frame"),
    56	    robot_cfg: SceneEntityCfg = SceneEntityCfg("robot"),
    57	) -> torch.Tensor:
    58	    objects_placed = torch.ones(env.num_envs, dtype=torch.bool, device=env.device)
    59	    container = env.scene[container_cfg.name]
    60	    ee_frame = env.scene[ee_frame_cfg.name]
    61	    robot = env.scene[robot_cfg.name]
    62	    env_origins = robot.data.root_pos_w
    63	    robot_quat = robot.data.root_quat_w
    64	    container_position = helpers.get_robot_relative_position(
    65	        container.data.root_pos_w - env_origins, robot_quat
    66	    )
    67	    containier_size = helpers.get_object_relative_bbox(
    68	        container_size, container.data.root_quat_w, robot_quat
    69	    )
    70	    for obj in objects:
    71	        object = env.scene[obj]
    72	        object_position = helpers.get_robot_relative_position(
    73	            object.data.root_pos_w - env_origins, robot_quat
    74	        )
    75	        object_size = helpers.get_object_relative_bbox(
    76	            object_sizes[obj], object.data.root_quat_w, robot_quat
    77	        )
    78	        objects_placed = torch.logical_and(
    79	            objects_placed,
    80	            helpers.is_object_placed(
    81	                object_position,
    82	                object_size,
    83	                container_position,
    84	                containier_size,
    85	            ),
    86	        )
    87	
    88	    goal_position_r = goal_position.to(device=env_origins.device)
    89	    eef_position_r = helpers.get_robot_relative_position(
    90	        ee_frame.data.target_pos_w[..., 0, :] - env_origins, robot_quat
    91	    )
    92	    eef_goal_dist = torch.norm(goal_position_r - eef_position_r, dim=1)
    93	    return torch.logical_and(objects_placed, eef_goal_dist < tolerance)
